Point the bot at an entity within five blocks before running this program. It reads an entity reference that includes the connection epoch, so SoulFire can reject it if the bot reconnects and a network ID is reused.

<Tabs items={['TypeScript', 'Python']} groupId="recipe-language">
  <Tab value="TypeScript">
    ```ts title="follow-entity.ts" lineNumbers
    import { SoulFire } from "@soulfiremc/sdk/promise";

    function env(name: string): string {
      const value = process.env[name];
      if (!value) throw new Error(`Set ${name} before running this recipe`);
      return value;
    }

    await using soulfire = await SoulFire.connect({
      baseUrl: env("SOULFIRE_URL"),
      token: env("SOULFIRE_TOKEN"),
    });

    const bot = soulfire
      .instance(env("SOULFIRE_INSTANCE_ID"))
      .bot(env("SOULFIRE_BOT_ID"));
    await bot.start();

    const target = await bot.world.entityAtCursor(5);
    if (!target?.reference) throw new Error("Point the bot at an entity first");

    const task = await bot.tasks.followEntity(target.reference, 3);
    console.log("Follow task ID:", task.id);
    ```
  </Tab>
  <Tab value="Python">
    ```python title="follow_entity.py" lineNumbers
    import asyncio
    import os

    from soulfire import AsyncSoulFire


    async def main() -> None:
        async with AsyncSoulFire.connect(
            os.environ["SOULFIRE_URL"],
            token=os.environ["SOULFIRE_TOKEN"],
        ) as soulfire:
            bot = soulfire.instance(os.environ["SOULFIRE_INSTANCE_ID"]).bot(
                os.environ["SOULFIRE_BOT_ID"]
            )
            await bot.start()

            target = await bot.world.entity_at_cursor(5)
            if target is None or not target.HasField("reference"):
                raise RuntimeError("Point the bot at an entity first")

            task = await bot.tasks.follow_entity(target.reference, distance=3)
            print("Follow task ID:", task.id)


    asyncio.run(main())
    ```
  </Tab>
</Tabs>

The server owns the follow task after it starts. Use [task controls](/docs/sdk/tasks) to watch it, stop it, or reconnect to it by ID.
