Point the bot at an entity within five blocks before running this program. It reads an entity reference that includes the connection epoch, so SoulFire can reject it if the bot reconnects and a network ID is reused.

<Tabs items={['TypeScript', 'Python']} groupId="recipe-language">
  <Tab value="TypeScript">
    ```ts title="follow-entity.ts" doc-test lineNumbers
    import { Effect } from "effect";
    import { SoulFire } from "@soulfiremc/sdk";

    function env(name: string): string {
      const value = process.env[name];
      if (!value) throw new Error(`Set ${name} before running this recipe`);
      return value;
    }

    const program = Effect.scoped(
      Effect.gen(function* () {
        const soulfire = yield* SoulFire.connect({
          baseUrl: env("SOULFIRE_URL"),
          token: env("SOULFIRE_TOKEN"),
        });

        const bot = soulfire
          .instance(env("SOULFIRE_INSTANCE_ID"))
          .bot(env("SOULFIRE_BOT_ID"));
        yield* bot.start();
        yield* bot.waitForOnline();

        const target = yield* bot.world.entityAtCursor(5);
        if (!target?.reference) throw new Error("Point the bot at an entity first");

        const task = yield* bot.tasks.followEntity(target.reference, 3);
        console.log("Follow task ID:", task.id);
      }),
    );

    await Effect.runPromise(program);
    ```
  </Tab>
  <Tab value="Python">
    ```python title="follow_entity.py" doc-test lineNumbers
    import asyncio
    import os

    from effect_py import EffectGen, Scope, gen, run_async, scoped
    from soulfire import SoulFireOperationError, SoulFire


    @gen
    def program() -> EffectGen[None, SoulFireOperationError, Scope]:
        soulfire = yield from SoulFire.connect(
            os.environ["SOULFIRE_URL"],
            token=os.environ["SOULFIRE_TOKEN"],
        )
        bot = soulfire.instance(os.environ["SOULFIRE_INSTANCE_ID"]).bot(
            os.environ["SOULFIRE_BOT_ID"]
        )
        yield from bot.start()
        yield from bot.wait_for_online()

        target = yield from bot.world.entity_at_cursor(5)
        if target is None or not target.HasField("reference"):
            raise RuntimeError("Point the bot at an entity first")

        task = yield from bot.tasks.follow_entity(target.reference, distance=3)
        print("Follow task ID:", task.id)


    asyncio.run(run_async(scoped(program).or_die()))
    ```
  </Tab>
</Tabs>

This recipe uses `start` rather than scoped `connect` so the bot and detached follow task stay running after the program exits.
The server owns the follow task after it starts. Use [task controls](/docs/sdk/tasks) to watch it, stop it, or reconnect to it by ID.
