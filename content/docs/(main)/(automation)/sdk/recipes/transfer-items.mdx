Set the chest coordinates in the program. The bot must be near a loaded, accessible container. The scoped handle closes the menu when the program leaves the scope, including when a transfer fails.

<Tabs items={['TypeScript', 'Python']} groupId="recipe-language">
  <Tab value="TypeScript">
    ```ts title="transfer-items.ts" doc-test lineNumbers
    import { Effect } from "effect";
    import { SoulFire, itemSelector } from "@soulfiremc/sdk";

    function env(name: string): string {
      const value = process.env[name];
      if (!value) throw new Error(`Set ${name} before running this recipe`);
      return value;
    }

    const program = Effect.scoped(
      Effect.gen(function* () {
        const soulfire = yield* SoulFire.connect({
          baseUrl: env("SOULFIRE_URL"),
          token: env("SOULFIRE_TOKEN"),
        });

        const bot = soulfire
          .instance(env("SOULFIRE_INSTANCE_ID"))
          .bot(env("SOULFIRE_BOT_ID"));
        yield* bot.connect();

        const chestPosition = {
          dimension: "minecraft:overworld",
          x: 10,
          y: 64,
          z: 10,
        };

        const chest = yield* bot.inventory.openScoped(chestPosition);
        yield* chest.deposit(itemSelector("cobblestone"), 64);
        yield* chest.withdraw(itemSelector("#foods"), 16);
      }),
    );

    await Effect.runPromise(program);
    ```

  <Tab value="Python">
    ```python title="transfer_items.py" doc-test lineNumbers
    import asyncio
    import os

    from effect_py import EffectGen, Scope, gen, run_async, scoped
    from soulfire import SoulFireOperationError, SoulFire, BlockPosition, item_selector


    @gen
    def program() -> EffectGen[None, SoulFireOperationError, Scope]:
        soulfire = yield from SoulFire.connect(
            os.environ["SOULFIRE_URL"],
            token=os.environ["SOULFIRE_TOKEN"],
        )
        bot = soulfire.instance(os.environ["SOULFIRE_INSTANCE_ID"]).bot(
            os.environ["SOULFIRE_BOT_ID"]
        )
        yield from bot.connect()

        chest_position = BlockPosition(
            dimension="minecraft:overworld", x=10, y=64, z=10
        )
        chest = yield from bot.inventory.open(chest_position)
        yield from chest.deposit(
            item_selector("cobblestone"), 64
        )
        yield from chest.withdraw(item_selector("#foods"), 16)


    asyncio.run(run_async(scoped(program).or_die()))
    ```
  </Tab>
