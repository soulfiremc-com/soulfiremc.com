This program collects up to 16 oak logs near the bot. The task runs on the SoulFire server. Save `task.id` if another process needs to find it later.

<Tabs items={['TypeScript', 'Python']} groupId="recipe-language">
  <Tab value="TypeScript">
    ```ts title="collect-blocks.ts" doc-test lineNumbers
    import { Effect, Stream } from "effect";
    import { SoulFire } from "@soulfiremc/sdk";

    function env(name: string): string {
      const value = process.env[name];
      if (!value) throw new Error(`Set ${name} before running this recipe`);
      return value;
    }

    const program = Effect.scoped(
      Effect.gen(function* () {
        const soulfire = yield* SoulFire.connect({
          baseUrl: env("SOULFIRE_URL"),
          token: env("SOULFIRE_TOKEN"),
        });

        const bot = soulfire
          .instance(env("SOULFIRE_INSTANCE_ID"))
          .bot(env("SOULFIRE_BOT_ID"));
        yield* bot.start();
        yield* bot.waitForOnline();

        const task = yield* bot.tasks.collectBlocks(["minecraft:oak_log"], {
          count: 16,
          searchRadius: 32,
          idempotencyKey: crypto.randomUUID(),
        });

        console.log("Task ID:", task.id);
        const result = yield* task.result();
        console.log(result);
      }),
    );

    await Effect.runPromise(program);
    ```
  </Tab>
  <Tab value="Python">
    ```python title="collect_blocks.py" doc-test lineNumbers
    import asyncio
    import os
    from uuid import uuid4

    from effect_py import EffectGen, Scope, gen, run_async, scoped
    from soulfire import SoulFireOperationError, SoulFire


    @gen
    def program() -> EffectGen[None, SoulFireOperationError, Scope]:
        soulfire = yield from SoulFire.connect(
            os.environ["SOULFIRE_URL"],
            token=os.environ["SOULFIRE_TOKEN"],
        )
        bot = soulfire.instance(os.environ["SOULFIRE_INSTANCE_ID"]).bot(
            os.environ["SOULFIRE_BOT_ID"]
        )
        yield from bot.start()
        yield from bot.wait_for_online()

        task = yield from bot.tasks.collect_blocks(
            ["minecraft:oak_log"],
            count=16,
            search_radius=32,
            idempotency_key=str(uuid4()),
        )

        print("Task ID:", task.id)
        result = yield from task.result()
        print(result)


    asyncio.run(run_async(scoped(program).or_die()))
    ```
  </Tab>
</Tabs>

The bot needs loaded chunks and suitable tools. If the task cannot reach every block, inspect its result and [task progress](/docs/sdk/tasks). The idempotency key prevents a duplicate task when the start request is retried.
